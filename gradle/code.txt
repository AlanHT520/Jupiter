package net.alan.jupiter.domain.enums;

public enum CheckStatus {

    PRESENT("在寝", 1),
    PERSONAL_LEAVE("事假", 2),
    SICK_LEAVE("病假", 3),
    ABSENT("旷宿", 4),
    MILITARY("征兵", 5),
    INTERN("实习", 6),
    DAY_STUDENT("走读", 7),
    LATE_RETURN("晚归", 8),
    LONG_SICK_LEAVE("长期病假", 9),
    UNKNOWN("未知", 0);

    private final String label;
    private final int code;

    CheckStatus(String label, int code) {
        this.label = label;
        this.code = code;
    }

    public String getLabel() {
        return label;
    }

    public int getCode() {
        return code;
    }

    public boolean isLeave() {
        return this == PERSONAL_LEAVE || this == SICK_LEAVE;
    }

    public static CheckStatus fromCode(int code) {
        for (CheckStatus status : values()) {
            if (status.code == code) {
                return status;
            }
        }
        return UNKNOWN;
    }
}

package net.alan.jupiter.infrastructure.excel;

import lombok.extern.slf4j.Slf4j;
import net.alan.jupiter.common.exception.BizException;
import net.alan.jupiter.common.exception.ErrorCode;
import net.alan.jupiter.domain.model.Student;
import org.apache.poi.ss.usermodel.*;
import org.apache.poi.ss.util.NumberToTextConverter;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;
import org.springframework.stereotype.Component;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.util.*;

@Slf4j @Component
public class ExcelImportHelper {

    /* 解析班级花名册 Excel。
       支持模板列序：序号 | 班级 | 姓名 | 学号 | 性别 | 宿舍楼 | 宿舍号 | 征兵 | 实习 | 请假
       也兼容旧模板：学号 | 姓名 | 性别 | 班级 | 楼栋 | 楼层 | 房间 | 床位 */
    public List<Student> parseClassRoster(byte[] excelData) {
        try (Workbook workbook = new XSSFWorkbook(new ByteArrayInputStream(excelData))) {
            Sheet sheet = workbook.getSheetAt(0);
            if (sheet == null) {
                throw new BizException(ErrorCode.EXCEL_FORMAT_ERROR, "Excel文件中未找到工作表");
            }
            int headerRowIdx = findHeaderRow(sheet);
            ColumnMapping mapping = detectColumnMapping(sheet, headerRowIdx);
            List<Student> students = new ArrayList<>();
            for (int i = headerRowIdx + 1; i <= sheet.getLastRowNum(); i++) {
                Row row = sheet.getRow(i);
                if (row == null) continue;
                try {
                    String studentNo = mapping.studentNoIdx >= 0
                            ? getCellString(row, mapping.studentNoIdx) : null;
                    if (studentNo == null || studentNo.isBlank()) continue;
                    Student student = Student.builder()
                            .studentNo(studentNo)
                            .name(mapping.nameIdx >= 0 ? getCellString(row, mapping.nameIdx) : null)
                            .gender(mapping.genderIdx >= 0 ? getCellString(row, mapping.genderIdx) : null)
                            .className(mapping.classNameIdx >= 0 ? getCellString(row, mapping.classNameIdx) : null)
                            .buildingName(mapping.buildingIdx >= 0 ? getCellString(row, mapping.buildingIdx) : null)
                            .floorNumber(mapping.floorIdx >= 0
                                    ? parseInteger(getCellString(row, mapping.floorIdx)) : null)
                            .roomNumber(mapping.roomIdx >= 0 ? getCellString(row, mapping.roomIdx) : null)
                            .bedNumber(mapping.bedIdx >= 0 ? getCellString(row, mapping.bedIdx) : null)
                            .phone(mapping.phoneIdx >= 0 ? getCellString(row, mapping.phoneIdx) : null)
                            .build();
                    students.add(student);
                } catch (Exception e) {
                    log.warn("解析花名册第{}行失败: {}", i + 1, e.getMessage());
                }
            }
            return students;
        } catch (IOException e) {
            throw new BizException(ErrorCode.EXCEL_FORMAT_ERROR, "无法读取Excel文件");
        }
    }

    private int findHeaderRow(Sheet sheet) {
        for (int rowIdx = 0; rowIdx <= Math.min(sheet.getLastRowNum(), 5); rowIdx++) {
            Row row = sheet.getRow(rowIdx);
            if (row == null) continue;
            for (int colIdx = 0; colIdx <= row.getLastCellNum(); colIdx++) {
                Cell cell = row.getCell(colIdx);
                if (cell != null && "学号".equals(getCellStringDirect(cell))) {
                    return rowIdx;
                }
            }
        }
        throw new BizException(ErrorCode.EXCEL_FORMAT_ERROR,
                "未找到表头行，请确保 Excel 包含「学号」列");
    }

    private ColumnMapping detectColumnMapping(Sheet sheet, int headerRowIdx) {
        Row headerRow = sheet.getRow(headerRowIdx);
        ColumnMapping mapping = new ColumnMapping();
        for (int colIdx = 0; colIdx <= headerRow.getLastCellNum(); colIdx++) {
            String header = getCellString(headerRow, colIdx);
            if (header == null) continue;
            switch (header) {
                case "学号" -> mapping.studentNoIdx = colIdx;
                case "姓名" -> mapping.nameIdx = colIdx;
                case "性别" -> mapping.genderIdx = colIdx;
                case "班级" -> mapping.classNameIdx = colIdx;
                case "楼栋", "宿舍楼", "宿舍楼编号" -> mapping.buildingIdx = colIdx;
                case "楼层" -> mapping.floorIdx = colIdx;
                case "房间", "宿舍号", "房间号" -> mapping.roomIdx = colIdx;
                case "床位", "床位号" -> mapping.bedIdx = colIdx;
                case "手机号" -> mapping.phoneIdx = colIdx;
            }
        }
        if (mapping.studentNoIdx < 0 || mapping.nameIdx < 0) {
            throw new BizException(ErrorCode.EXCEL_FORMAT_ERROR,
                    "表头必须包含「学号」和「姓名」列");
        }
        return mapping;
    }

    private static class ColumnMapping {
        int studentNoIdx = -1, nameIdx = -1, genderIdx = -1, classNameIdx = -1;
        int buildingIdx = -1, floorIdx = -1, roomIdx = -1, bedIdx = -1, phoneIdx = -1;
    }

    private String getCellStringDirect(Cell cell) {
        if (cell == null) return "";
        return switch (cell.getCellType()) {
            case STRING -> cell.getStringCellValue().trim();
            case NUMERIC -> {
                if (DateUtil.isCellDateFormatted(cell)) {
                    try { yield cell.getLocalDateTimeCellValue().toLocalDate().toString(); }
                    catch (Exception e) { yield NumberToTextConverter.toText(cell.getNumericCellValue()); }
                }
                yield NumberToTextConverter.toText(cell.getNumericCellValue());
            }
            case BOOLEAN -> String.valueOf(cell.getBooleanCellValue());
            case FORMULA -> {
                try { yield cell.getStringCellValue().trim(); }
                catch (Exception e1) {
                    try { yield NumberToTextConverter.toText(cell.getNumericCellValue()); }
                    catch (Exception e2) { yield cell.getCellFormula(); }
                }
            }
            case BLANK -> "";
            default -> "";
        };
    }

    private Integer parseInteger(String value) {
        if (value == null || value.isBlank()) return null;
        try { return (int) Double.parseDouble(value); }
        catch (NumberFormatException e) { return null; }
    }

    @lombok.Data
    public static class RoomAssignmentRow {
        private String studentNo;
        private Integer floorNumber;
        private String roomNumber;
        private String bedNumber;
        private String remark;
    }

    public List<RoomAssignmentRow> parseRoomAssignment(byte[] excelData) {
        List<RoomAssignmentRow> rows = new ArrayList<>();
        try (Workbook workbook = new XSSFWorkbook(new ByteArrayInputStream(excelData))) {
            Sheet sheet = workbook.getSheetAt(0);
            if (sheet == null) throw new BizException(ErrorCode.EXCEL_FORMAT_ERROR, "Excel文件中未找到工作表");
            Row headerRow = sheet.getRow(0);
            if (headerRow == null) throw new BizException(ErrorCode.EXCEL_FORMAT_ERROR, "Excel文件缺少标题行");
            for (int i = 1; i <= sheet.getLastRowNum(); i++) {
                Row row = sheet.getRow(i);
                if (row == null) continue;
                try {
                    RoomAssignmentRow rar = new RoomAssignmentRow();
                    rar.setStudentNo(getCellString(row, 0));
                    rar.setFloorNumber(parseInteger(getCellString(row, 1)));
                    rar.setRoomNumber(getCellString(row, 2));
                    rar.setBedNumber(getCellString(row, 3));
                    rar.setRemark(getCellString(row, 4));
                    rows.add(rar);
                } catch (Exception e) {
                    log.warn("解析住宿表第{}行失败: {}", i + 1, e.getMessage());
                }
            }
        } catch (IOException e) {
            throw new BizException(ErrorCode.EXCEL_FORMAT_ERROR, "无法读取Excel文件");
        }
        return rows;
    }

    private String getCellString(Row row, int index) {
        Cell cell = row.getCell(index);
        if (cell == null) return null;
        return switch (cell.getCellType()) {
            case STRING -> cell.getStringCellValue().trim();
            case NUMERIC -> {
                if (DateUtil.isCellDateFormatted(cell)) {
                    try { yield cell.getLocalDateTimeCellValue().toLocalDate().toString(); }
                    catch (Exception e) { yield NumberToTextConverter.toText(cell.getNumericCellValue()); }
                }
                yield NumberToTextConverter.toText(cell.getNumericCellValue());
            }
            case BOOLEAN -> String.valueOf(cell.getBooleanCellValue());
            case FORMULA -> {
                try { yield cell.getStringCellValue().trim(); }
                catch (Exception e1) {
                    try { yield NumberToTextConverter.toText(cell.getNumericCellValue()); }
                    catch (Exception e2) { yield cell.getCellFormula(); }
                }
            }
            case BLANK -> null;
            default -> null;
        };
    }
}

    public byte[] exportClassReport(List<ClassReportExportRow> rows, String className, String buildingName) {
        try (Workbook workbook = new XSSFWorkbook()) {
            Sheet sheet = workbook.createSheet(className != null ? className : "班级查宿汇总");
            CellStyle headerStyle = createHeaderStyle(workbook);
            Row headerRow = sheet.createRow(0);
            for (int i = 0; i < CLASS_REPORT_HEADERS.length; i++) {
                Cell cell = headerRow.createCell(i);
                cell.setCellValue(CLASS_REPORT_HEADERS[i]);
                cell.setCellStyle(headerStyle);
            }
            int rowIdx = 1;
            for (ClassReportExportRow r : rows) {
                Row row = sheet.createRow(rowIdx++);
                row.createCell(0).setCellValue(r.getStudentNo() != null ? r.getStudentNo() : "");
                row.createCell(1).setCellValue(r.getStudentName() != null ? r.getStudentName() : "");
                row.createCell(2).setCellValue(r.getGender() != null ? r.getGender() : "");
                row.createCell(3).setCellValue(buildingName != null ? buildingName : "");
                row.createCell(4).setCellValue(r.getFloorNumber() != null ? String.valueOf(r.getFloorNumber()) : "");
                row.createCell(5).setCellValue(r.getRoomNumber() != null ? r.getRoomNumber() : "");
                row.createCell(6).setCellValue(r.getCheckStatus() != null ? r.getCheckStatus() : "");
                row.createCell(7).setCellValue(r.getLeaveType() != null ? r.getLeaveType() : "");
                row.createCell(8).setCellValue(r.isHasProof() ? "有" : "");
                row.createCell(9).setCellValue(r.getRemark() != null ? r.getRemark() : "");
            }
            autoSizeColumns(sheet, CLASS_REPORT_HEADERS.length);
            return toByteArray(workbook);
        } catch (IOException e) {
            throw new BizException(ErrorCode.EXPORT_ERROR, "导出班级查宿汇总失败: " + e.getMessage());
        }

    public byte[] exportCheckRecords(List<CheckRecord> records, String title) {
        try (Workbook workbook = new XSSFWorkbook()) {
            Sheet sheet = workbook.createSheet(title != null ? title : "查寝记录");
            CellStyle headerStyle = createHeaderStyle(workbook);
            Row headerRow = sheet.createRow(0);
            String[] headers = {"学号", "姓名", "房间号", "查寝状态", "状态来源", "备注", "提交人", "提交时间"};
            for (int i = 0; i < headers.length; i++) {
                Cell cell = headerRow.createCell(i);
                cell.setCellValue(headers[i]);
                cell.setCellStyle(headerStyle);
            }
            int rowIdx = 1;
            for (CheckRecord record : records) {
                Row row = sheet.createRow(rowIdx++);
                row.createCell(0).setCellValue(record.getStudentNo() != null ? record.getStudentNo() : "");
                row.createCell(1).setCellValue(record.getStudentName() != null ? record.getStudentName() : "");
                row.createCell(2).setCellValue(record.getRoomNumber() != null ? record.getRoomNumber() : "");

                row.createCell(7).setCellValue(record.getSubmittedAt() != null
                        ? record.getSubmittedAt().toString() : "");

            autoSizeColumns(sheet, headers.length);

        } catch (IOException e) {
            throw new BizException(ErrorCode.EXPORT_ERROR, "导出Excel失败: " + e.getMessage());

    private CellStyle createHeaderStyle(Workbook workbook) {
        CellStyle style = workbook.createCellStyle();
        Font font = workbook.createFont();
        font.setBold(true);
        style.setFont(font);
        style.setFillForegroundColor(IndexedColors.GREY_25_PERCENT.getIndex());
        style.setFillPattern(FillPatternType.SOLID_FOREGROUND);
        style.setBorderBottom(BorderStyle.THIN);
        style.setBorderTop(BorderStyle.THIN);
        style.setBorderLeft(BorderStyle.THIN);
        style.setBorderRight(BorderStyle.THIN);
        return style;
    }

    private void autoSizeColumns(Sheet sheet, int columnCount) {
        for (int i = 0; i < columnCount; i++) {
            sheet.autoSizeColumn(i);
            int width = Math.min(sheet.getColumnWidth(i) + 512, 15 * 256);
            sheet.setColumnWidth(i, width);

package net.alan.jupiter.application.roster;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;

import org.springframework.transaction.annotation.Transactional;
import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.util.*;
import java.util.stream.Collectors;

@Slf4j @Service @RequiredArgsConstructor
public class ImportClassRosterService {

    private static final List<String> TEMPLATE_HEADERS = List.of(
            "班级", "姓名", "性别", "学号", "宿舍楼编号", "楼层", "房间号", "床位号", "手机号", "备注"
    );

    private final StudentRepository studentRepository;

    private final FloorRepository floorRepository;
    private final ExcelImportHelper excelImportHelper;

    public byte[] downloadTemplate() {
        try (Workbook workbook = new XSSFWorkbook()) {
            Sheet sheet = workbook.createSheet("花名册");
            Row headerRow = sheet.createRow(0);
            for (int i = 0; i < TEMPLATE_HEADERS.size(); i++) {
                headerRow.createCell(i).setCellValue(TEMPLATE_HEADERS.get(i));
            }
            ByteArrayOutputStream bos = new ByteArrayOutputStream();
            workbook.write(bos);

        } catch (IOException e) {
            throw new BizException(ErrorCode.INTERNAL_ERROR, "生成模板失败");
        }
    }

    public List<Student> previewImport(byte[] excelData) {
        return excelImportHelper.parseClassRoster(excelData);
    }

    @Transactional
    public ImportResult importClassRoster(byte[] excelData, String fileName) {
        List<Student> students = excelImportHelper.parseClassRoster(excelData);
        ImportResult result = new ImportResult();
        List<Student> toSave = new ArrayList<>();
        List<String> allStudentNos = students.stream()
                .map(Student::getStudentNo).filter(Objects::nonNull).toList();
        Set<String> existingStudentNos = studentRepository.findByStudentNoIn(allStudentNos)
                .stream().map(Student::getStudentNo).collect(Collectors.toSet());
        Map<String, ClassInfo> classInfoCache = new HashMap<>();
        Map<String, DormBuilding> buildingCache = new HashMap<>();
        Map<String, Floor> floorCache = new HashMap<>();
        Map<String, Room> roomCache = buildRoomCache();

        for (Student student : students) {
            if (student.getStudentNo() == null || student.getStudentNo().isBlank()) {
                result.addError("学号为空"); continue;
            }
            if (existingStudentNos.contains(student.getStudentNo())) {
                Student existing = studentRepository.findByStudentNo(student.getStudentNo()).orElse(null);
                if (existing != null) {
                    updateExistingStudent(existing, student, classInfoCache, buildingCache, floorCache, roomCache);
                    studentRepository.save(existing);
                    result.addUpdated(student.getStudentNo());
                } else {
                    result.addSkipped(student.getStudentNo());
                }
                continue;
            }
            resolveAssociations(student, classInfoCache, buildingCache, floorCache, roomCache);
            toSave.add(student);
        }
        studentRepository.saveAll(toSave);
        result.setImported(toSave.size());
        return result;
    }

    private Map<String, Room> buildRoomCache() {
        return roomRepository.findAll().stream()
                .collect(Collectors.toMap(r -> {
                    String bName = r.getBuilding() != null ? r.getBuilding().getName() : "";
                    String fNum = r.getFloor() != null ? String.valueOf(r.getFloor().getFloorNumber()) : "";
                    return bName + "|" + fNum + "|" + r.getRoomNumber();
                }, r -> r, (a, b) -> a));
    }

    private void resolveAssociations(Student student,
                                     Map<String, ClassInfo> classInfoCache,
                                     Map<String, DormBuilding> buildingCache,
                                     Map<String, Floor> floorCache,
                                     Map<String, Room> roomCache) {
        resolveClass(student, classInfoCache);
        resolveBuilding(student, buildingCache);
        resolveFloorAndRoom(student, buildingCache, floorCache, roomCache);
    }

    private void resolveClass(Student student, Map<String, ClassInfo> classInfoCache) {
        String className = student.getClassName();
        if (className == null || className.isBlank()) return;
        ClassInfo classInfo = classInfoCache.computeIfAbsent(className, cn -> {
            ClassInfo ci = classInfoRepository.findByName(cn).orElse(null);
            if (ci == null) { ci = new ClassInfo(); ci.setName(cn); ci = classInfoRepository.save(ci); }
            return ci;
        });
        student.setClassInfo(classInfo);
        student.setClassName(classInfo.getName());
    }

    private void resolveBuilding(Student student, Map<String, DormBuilding> buildingCache) {
        String buildingName = student.getBuildingName();
        if (buildingName == null || buildingName.isBlank()) return;
        DormBuilding building = buildingCache.computeIfAbsent(buildingName, bn -> {
            DormBuilding b = buildingRepository.findByName(bn).orElse(null);
            if (b == null) { b = new DormBuilding(); b.setName(bn); b.setBuildingCode(bn); b.setIsActive(true); b = buildingRepository.save(b); }
            return b;
        });
        student.setBuilding(building);
        student.setBuildingName(building.getName());
    }

    private void resolveFloorAndRoom(Student student,
                                     Map<String, DormBuilding> buildingCache,
                                     Map<String, Floor> floorCache,
                                     Map<String, Room> roomCache) {
        String buildingName = student.getBuildingName();
        String roomNumber = student.getRoomNumber();
        if (buildingName == null || buildingName.isBlank()) return;
        DormBuilding building = buildingCache.get(buildingName);
        if (building == null) return;

        String floorKey = building.getName() + "|"
                + (student.getFloorNumber() != null ? student.getFloorNumber() : 0);
        Floor floor = floorCache.computeIfAbsent(floorKey, fk -> {
            Floor f = floorRepository
                    .findByBuildingIdAndFloorNumber(building.getId(), student.getFloorNumber())
                    .orElse(null);
            if (f == null) {
                f = new Floor(); f.setBuilding(building);
                f.setFloorNumber(student.getFloorNumber());
                f.setFloorName((student.getFloorNumber() != null ? student.getFloorNumber() : 0) + "层");
                f = floorRepository.save(f);
            }
            return f;
        });

        if (roomNumber == null || roomNumber.isBlank()) return;

        String roomKey = building.getName() + "|"
                + (floor.getFloorNumber() != null ? floor.getFloorNumber() : 0) + "|" + roomNumber;
        Room room = roomCache.computeIfAbsent(roomKey, rk -> {
            Room r = roomRepository.findByRoomNumberAndBuildingId(roomNumber, building.getId()).orElse(null);
            if (r == null) {
                r = new Room(); r.setBuilding(building); r.setRoomNumber(roomNumber);
                r.setFloor(floor); r.setIsActive(true); r = roomRepository.save(r);
            }
            return r;
        });
        student.setRoom(room);
        student.setRoomNumber(room.getRoomNumber());
        student.setFloorNumber(floor.getFloorNumber());
    }

    private void updateExistingStudent(Student existing, Student incoming,
                                       Map<String, ClassInfo> classInfoCache,
                                       Map<String, DormBuilding> buildingCache,
                                       Map<String, Floor> floorCache,
                                       Map<String, Room> roomCache) {
        if (incoming.getName() != null && !incoming.getName().isBlank()) existing.setName(incoming.getName());
        if (incoming.getGender() != null && !incoming.getGender().isBlank()) existing.setGender(incoming.getGender());
        if (incoming.getClassName() != null && !incoming.getClassName().isBlank()) {
            existing.setClassName(incoming.getClassName());
            resolveClass(existing, classInfoCache);
        }
        if (incoming.getBuildingName() != null && !incoming.getBuildingName().isBlank()) {
            existing.setBuildingName(incoming.getBuildingName());
            resolveBuilding(existing, buildingCache);
        }
        if (incoming.getFloorNumber() != null) existing.setFloorNumber(incoming.getFloorNumber());
        if (incoming.getRoomNumber() != null && !incoming.getRoomNumber().isBlank()) {
            existing.setRoomNumber(incoming.getRoomNumber());
            resolveFloorAndRoom(existing, buildingCache, floorCache, roomCache);
        }
        if (incoming.getBedNumber() != null) existing.setBedNumber(incoming.getBedNumber());
        if (incoming.getPhone() != null) existing.setPhone(incoming.getPhone());
        if (incoming.getRemark() != null) existing.setRemark(incoming.getRemark());
    }

    @lombok.Data
    public static class ImportResult {
        private int imported, updated, skipped;
        private final List<String> skippedNos = new ArrayList<>();
        private final List<String> updatedNos = new ArrayList<>();
        private final List<String> errors = new ArrayList<>();
        public void addSkipped(String studentNo) { skipped++; skippedNos.add(studentNo); }
        public void addUpdated(String studentNo) { updated++; updatedNos.add(studentNo); }
        public void addError(String message) { errors.add(message); }
    }

    private NewStudentStats calcNewStudentStats(List<DormBuilding> buildings,
                                                  Map<String, CheckRecord> recordMap) {
        List<Student> allStudents = new ArrayList<>();
        for (DormBuilding b : buildings) allStudents.addAll(studentRepository.findByBuildingId(b.getId()));
        int should = 0, actual = 0, male = 0, female = 0, militaryMale = 0, leave = 0;
        for (Student s : allStudents) {
            if (s.getIsNewStudent() == null || !s.getIsNewStudent()) continue;
            if (!isActive(s)) continue;
            CheckStatus status = resolveStatus(s, recordMap);
            should++;
            if ("男".equals(s.getGender())) male++;
            else if ("女".equals(s.getGender())) female++;
            switch (status) {
                case PRESENT -> actual++;
                case PERSONAL_LEAVE, SICK_LEAVE -> leave++;
                case MILITARY -> { if ("男".equals(s.getGender())) militaryMale++; }
            }
        }
        var ns = new NewStudentStats();
        ns.setShould(should); ns.setActual(actual);
        ns.setMale(male); ns.setFemale(female);
        ns.setMilitaryMale(militaryMale); ns.setLeave(leave);
        return ns;
    }

    private Map<String, CheckRecord> buildRecordMap(List<CheckRecord> records) {
        return records.stream()
                .filter(r -> r.getStudent() != null)
                .collect(Collectors.toMap(r -> r.getStudent().getId(), r -> r, (a, b) -> a));
    }

    private Map<String, List<Student>> groupByRoomId(List<Student> students) {
        return students.stream()
                .collect(Collectors.groupingBy(s -> s.getRoom() != null ? s.getRoom().getId() : ""));
    }

    private boolean isActive(Student s) { return s.getIsActive() == null || s.getIsActive(); }

    private CheckStatus resolveStatus(Student student, Map<String, CheckRecord> recordMap) {
        CheckRecord record = recordMap.get(student.getId());
        if (record != null && record.getCheckStatus() != null) return record.getCheckStatus();
        CheckStatus preset = checkRuleService.determinePresetStatus(student);
        if (preset != null) return preset;
        return CheckStatus.UNKNOWN;
    }

    private String getRemark(Map<String, CheckRecord> recordMap, Student student) {
        CheckRecord record = recordMap.get(student.getId());
        return record != null && record.getRemark() != null ? record.getRemark() : "";
    }

    private boolean hasProof(Map<String, CheckRecord> recordMap, Student student) {
        CheckRecord record = recordMap.get(student.getId());
        return record != null && record.getProofFile() != null;
    }

    private String leaveType(CheckStatus status) {
        return switch (status) {
            case PERSONAL_LEAVE -> "事假";
            case SICK_LEAVE -> "病假";
            case LONG_SICK_LEAVE -> "长期病假";
            default -> "";
        };
    }

package net.alan.jupiter.fx.controller;

import javafx.application.Platform;
import javafx.fxml.FXML;
import javafx.scene.control.*;
import javafx.stage.FileChooser;
import lombok.RequiredArgsConstructor;
import net.alan.jupiter.application.report.ExportReportService;
import net.alan.jupiter.application.roster.ImportClassRosterService;
import net.alan.jupiter.fx.util.DialogUtils;
import net.alan.jupiter.fx.util.FxUtils;
import net.alan.jupiter.domain.repository.TaskRepository;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Sort;
import org.springframework.stereotype.Component;
import java.io.File;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;

@Component
@RequiredArgsConstructor
public class ImportExportController {

    private final ImportClassRosterService importClassRosterService;
    private final ExportReportService exportReportService;
    private final TaskRepository taskRepository;

    private static final FileChooser.ExtensionFilter EXCEL_FILTER =
            new FileChooser.ExtensionFilter("Excel 文件 (*.xlsx)", "*.xlsx");

    @FXML private TextArea logArea;
    @FXML private Label statusLabel;
    @FXML private Button importBtn;
    @FXML private Button exportBtn;

    @FXML
    public void initialize() {
        statusLabel.setText("就绪");
    }

    @FXML
    private void onImport() {
        FileChooser chooser = new FileChooser();
        chooser.setTitle("选择花名册文件");
        chooser.getExtensionFilters().add(EXCEL_FILTER);
        File file = chooser.showOpenDialog(importBtn.getScene().getWindow());
        if (file == null) return;

        statusLabel.setText("正在导入花名册");
        logArea.clear();

        FxUtils.runAsync(() -> {
            try {
                byte[] data = Files.readAllBytes(file.toPath());
                return importClassRosterService.importClassRoster(data, file.getName());
            } catch (IOException e) {
                throw new RuntimeException("读取文件失败: " + e.getMessage(), e);
            }
        }, result -> {
            appendLog("===== 花名册导入结果 =====");
            appendLog("文件名: " + file.getName());
            appendLog("导入: " + result.getImported() + " 条");
            appendLog("更新: " + result.getUpdated() + " 条");
            appendLog("跳过: " + result.getSkipped() + " 条");
            if (!result.getErrors().isEmpty()) {
                appendLog("错误: " + result.getErrors().size() + " 条");
                result.getErrors().forEach(e -> appendLog("  - " + e));
            }
            statusLabel.setText("导入完成: 导入" + result.getImported()
                    + ", 更新" + result.getUpdated() + ", 跳过" + result.getSkipped());
        });
    }

    @FXML
    private void onExport() {
        var taskPage = taskRepository.findAll(
                PageRequest.of(0, 100, Sort.by(Sort.Direction.DESC, "createdAt")));
        var tasks = taskPage.getContent();
        if (tasks.isEmpty()) {
            DialogUtils.showWarn("导出报表", "暂无查寝任务，请先分配任务");
            return;
        }

        ChoiceDialog<net.alan.jupiter.domain.model.Task> taskDialog = new ChoiceDialog<>(
                tasks.get(0), tasks);
        taskDialog.setTitle("选择任务");
        taskDialog.setHeaderText("请选择要导出报表的查寝任务");
        taskDialog.setContentText("任务:");

        taskDialog.showAndWait().ifPresent(task -> {
            FileChooser chooser = new FileChooser();
            chooser.setTitle("保存报表");
            chooser.getExtensionFilters().add(
                    new FileChooser.ExtensionFilter("HTML 文件 (*.html)", "*.html"));
            String defaultName = "report_" + task.getTaskNo() + "_"
                    + LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyyMMdd_HHmmss")) + ".html";
            chooser.setInitialFileName(defaultName);
            File file = chooser.showSaveDialog(exportBtn.getScene().getWindow());
            if (file == null) return;

            statusLabel.setText("正在导出报表");
            FxUtils.runAsync(() -> {
                String outputDir = file.getParent();
                String savedPath = exportReportService.exportToFile(
                        task.getId(), "building-report.ftl", outputDir);
                Path savedFile = Path.of(savedPath);
                if (!savedFile.equals(file.toPath())) {
                    try {
                        Files.move(savedFile, file.toPath(),
                                java.nio.file.StandardCopyOption.REPLACE_EXISTING);
                    } catch (IOException e) {
                        throw new RuntimeException(e);
                    }
                }
                return file.getAbsolutePath();
            }, savedPath -> {
                appendLog("===== 报表导出完成 =====");
                appendLog("任务编号: " + task.getTaskNo());
                appendLog("保存路径: " + savedPath);
                statusLabel.setText("报表已导出");
                DialogUtils.showInfo("导出成功", "报表已保存到:\n" + savedPath);
            });
        });
    }

    private void appendLog(String text) {
        Platform.runLater(() -> logArea.appendText(text + "\n"));
    }
}
